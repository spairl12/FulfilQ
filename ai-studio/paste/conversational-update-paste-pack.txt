Func<string, int, string> cut = (v, max) => v == null ? string.Empty : (v.Length <= max ? v : v.Substring(0, max));
Action<Guid, string, Guid, Guid, string, string, string, int> writeLedger =
	(lineId, type, productId, reasonId, checks, prior, next, sequence) => {
		Entity row = ledgerSchema.CreateEntity(uc);
		row.SetDefColumnValues();
		row.SetColumnValue("SPAIOpportunityId", opportunityId);
		row.SetColumnValue("SPAIScheduleLineId", lineId);
		row.SetColumnValue("SPAIDecisionTypeId", decisionType[type]);
		row.SetColumnValue("SPAIActor", "BP2a SPAIDeterministicMatch");
		if (productId != Guid.Empty) {
			row.SetColumnValue("SPAIProposedProductId", productId);
		}
		if (reasonId != Guid.Empty) {
			row.SetColumnValue("SPAIReasonCodeId", reasonId);
		}
		row.SetColumnValue("SPAIComplianceChecks", checks);
		row.SetColumnValue("SPAIPriorValue", cut(prior, 250));
		row.SetColumnValue("SPAINewValue", cut(next, 250));
		row.SetColumnValue("SPAISequence", sequence);
		row.SetColumnValue("SPAIOccurredOn", uc.CurrentUser.GetCurrentDateTime());
		row.Save(false);
	};

foreach (Entity line in lines) {
	int n = line.GetTypedColumnValue<int>(lc["SPAILineNumber"]);
	string model = (line.GetTypedColumnValue<string>(lc["SPAISpecifiedModel"]) ?? string.Empty).Trim();
	Entity p;
	if (model.Length == 0 || !products.TryGetValue(model, out p)) {
		unresolved++;
		continue;  // typo, or no model code at all: the Adjudicator's problem, not ours
	}
	string family = p.GetTypedColumnValue<string>(pc["SPAIProductFamily.Name"]) ?? string.Empty;
	bool[] r;
	if (!regimes.TryGetValue(family, out r)) {
		r = new[] { true, true, true, true };  // unknown family: every regime applies
	}
	int w = line.GetTypedColumnValue<int>(lc["SPAIRequiredCutoutW"]);
	int h = line.GetTypedColumnValue<int>(lc["SPAIRequiredCutoutH"]);
	int d = line.GetTypedColumnValue<int>(lc["SPAIRequiredCutoutD"]);
	// Each check records WHAT WAS FOUND, and "n/a" when the regime does not apply to the family, so an
	// auditor can tell "checked and passed" from "not applicable" (live run 2026-09-27 logged a wall
	// oven's WELS check as PASS, and a discontinued product as "fails: lifecycle Current").
	// The specified product IS the specification, so there is no rating comparison to make here.
	// Ratings are compared against it in BP3, where a DIFFERENT product is proposed.
	string familyText = family.Length > 0 ? family : "this product family";
	var checkParts = new List<string>();
	var failedParts = new List<string>();
	Action<string, bool, string, string> check = (rule, passed, found, failure) => {
		checkParts.Add(rule + ": " + (passed ? "PASS" : "FAIL") + " (" + (passed ? found : failure) + ")");
		if (!passed) {
			failedParts.Add(rule + " " + failure);
		}
	};
	Action<string, string> notApplicable = (rule, why) => checkParts.Add(rule + ": n/a (" + why + ")");
	Func<string, string> text = c => (p.GetTypedColumnValue<string>(pc[c]) ?? string.Empty).Trim();

	string lifecycle = text("SPAILifecycleStatus.Name");
	check("lifecycle", lifecycle == "Current", "Current",
		"is " + (lifecycle.Length > 0 ? lifecycle : "not set") + ", must be Current");
	check("project approval", p.GetTypedColumnValue<bool>(pc["SPAIProjectApproved"]), "approved",
		"not given");
	int pw = p.GetTypedColumnValue<int>(pc["SPAICutoutWidthMm"]);
	int ph = p.GetTypedColumnValue<int>(pc["SPAICutoutHeightMm"]);
	int pd = p.GetTypedColumnValue<int>(pc["SPAICutoutDepthMm"]);
	if (r[0]) {
		string productCut = string.Format("{0}x{1}x{2}", pw, ph, pd);
		check("cut-out", pw == w && ph == h && pd == d, productCut + " matches specified",
			string.Format("{0} does not match specified {1}x{2}x{3}", productCut, w, h, d));
	} else {
		notApplicable("cut-out", familyText + " has no cut-out rule");
	}
	string wels = text("SPAIWelsRegistrationNo");
	if (r[1]) {
		check("WELS", wels.Length > 0, "registered " + wels, "not registered");
	} else {
		notApplicable("WELS", familyText + " is not WELS-regulated");
	}
	string gems = text("SPAIGemsRegistrationNo");
	if (r[2]) {
		check("GEMS", gems.Length > 0, "registered " + gems, "not registered");
	} else {
		notApplicable("GEMS", familyText + " is not GEMS-regulated");
	}
	string waterMark = text("SPAIWaterMarkCertNo");
	if (r[3]) {
		check("WaterMark", waterMark.Length > 0, "certificate " + waterMark, "certificate missing");
	} else {
		notApplicable("WaterMark", familyText + " does not need WaterMark");
	}
	string checks = string.Join("; ", checkParts);
	Entity row = lineSchema.CreateEntity(uc);
	row.FetchFromDB(line.PrimaryColumnValue);
	if (failedParts.Count == 0) {
		decimal sell = p.GetTypedColumnValue<decimal>(pc["SPAITradeSellPrice"]);
		decimal cost = p.GetTypedColumnValue<decimal>(pc["SPAIWholesaleCost"]);
		int qty = line.GetTypedColumnValue<int>(lc["SPAIQuantity"]);
		row.SetColumnValue("SPAIMatchedProductId", p.PrimaryColumnValue);
		row.SetColumnValue("SPAIUnitCost", cost);
		row.SetColumnValue("SPAIUnitSell", sell);
		row.SetColumnValue("SPAILineTotal", sell * qty);
		row.SetColumnValue("SPAILineMarginPct", sell == 0m ? 0m : Math.Round((sell - cost) / sell * 100m, 1));
		row.SetColumnValue("SPAIResolvedBy", "Deterministic");
		row.Save(false);  // status stays Pending: BP2b decides Exact match vs multi-location vs shortfall
		writeLedger(line.PrimaryColumnValue, "Deterministic match", p.PrimaryColumnValue, Guid.Empty, checks,
			"Pending", "Matched " + p.GetTypedColumnValue<string>(pc["Code"]), n);
		matched++;
	} else {
		row.SetColumnValue("SPAIReasonCodeId", reasons["COMPLIANCE_FAIL"]);
		row.SetColumnValue("SPAIComplianceNotes", string.Join(" ", new[] {
			line.GetTypedColumnValue<string>(lc["SPAIComplianceNotes"]) ?? string.Empty,
			"Specified product fails the compliance floor: "
				+ string.Join("; ", failedParts) + "." }).Trim());
		row.Save(false);
		writeLedger(line.PrimaryColumnValue, "Compliance rejection", Guid.Empty, reasons["COMPLIANCE_FAIL"], checks,
			"Pending", "COMPLIANCE_FAIL on the specified product", n);
		floorFail++;
	}
}

totalEsq.Filters.Add(totalEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAIResolvedBy", "Deterministic"));

	opportunity.SetColumnValue("SPAIDeterministicCount", deterministicTotal);

	var statusByName = loadByName("SPAIAdjudicationStatus");
	Guid currentStatus = opportunity.GetTypedColumnValue<Guid>("SPAIAdjudicationStatusId");
	bool advanced = false;
	foreach (string later in new[] { "Awaiting Gate 2", "Submitted", "Awarded", "Re-adjudicating" }) {
		if (statusByName.ContainsKey(later) && currentStatus == statusByName[later]) {
			advanced = true;
		}
	}
	if (!advanced && matched + floorFail + unresolved > 0) {
		opportunity.SetColumnValue("SPAIAdjudicationStatusId", statusByName["Sourcing"]);
	}
	opportunity.Save(false);
}
Set("ProjectState", projectState);
Set("MatchedCount", matched);
Set("FloorFailCount", floorFail);
Set("UnresolvedCount", unresolved);
return true;

// ---- 2. Product loader (shared by specified-product and candidate queries) ----
var productColumns = new[] { "Code", "Name", "SPAIModelCode", "SPAIBrand.Name", "SPAIProductFamily.Name", "SPAILifecycleStatus.Name",
	"SPAIProjectApproved", "SPAICutoutWidthMm", "SPAICutoutHeightMm", "SPAICutoutDepthMm", "SPAIWelsRegistrationNo",

	"SPAILeadTimeWeeks", "SPAITradeSellPrice", "SPAIWholesaleCost", "SPAISupersededBy.Code" };
Func<Action<EntitySchemaQuery>, List<Dictionary<string, object>>> loadProducts = addFilters => {
	var esq = new EntitySchemaQuery(uc.EntitySchemaManager, "Product");
	esq.PrimaryQueryColumn.IsAlwaysSelect = true;
	var map = new Dictionary<string, string>();
	foreach (string c in productColumns) {
		map[c] = esq.AddColumn(c).Name;
	}
	addFilters(esq);
	var result = new List<Dictionary<string, object>>();
	foreach (Entity e in esq.GetEntityCollection(uc)) {
		var p = new Dictionary<string, object> { { "Id", e.PrimaryColumnValue } };
		foreach (var kv in map) {
			p[kv.Key] = e.GetColumnValue(kv.Value);
		}
		result.Add(p);
	}
	return result;
};
Func<Dictionary<string, object>, string, string> ps = (p, c) => p[c] == null ? string.Empty : p[c].ToString().Trim();
Func<Dictionary<string, object>, string, int> pi = (p, c) => p[c] == null ? 0 : Convert.ToInt32(p[c]);
Func<Dictionary<string, object>, string, decimal?> pd = (p, c) => p[c] == null || Convert.ToDecimal(p[c]) == 0m ? (decimal?)null : Convert.ToDecimal(p[c]);

Func<Entity, string> familyOf = line => {
	Dictionary<string, object> sp;
	if (specifiedByModel.TryGetValue(str(line, lc["SPAISpecifiedModel"]), out sp) && ps(sp, "SPAIProductFamily.Name").Length > 0) {
		return ps(sp, "SPAIProductFamily.Name");

	string haystack = str(line, lc["SPAIScheduleNotes"]) + " " + str(line, lc["SPAISpecifiedText"]);
	foreach (string f in familyNames) {
		if (haystack.IndexOf(f, StringComparison.OrdinalIgnoreCase) >= 0) {
			return f;
		}
	}
	return string.Empty;
};

			familyPool.Add(p);
		}
	}

	var eligible = new List<Dictionary<string, object>>();
	foreach (var p in familyPool) {
		bool fits = !r[0] || (pi(p, "SPAICutoutWidthMm") == w && pi(p, "SPAICutoutHeightMm") == h && pi(p, "SPAICutoutDepthMm") == d);
		if (!fits) {
			continue;
		}
		cutoutFit++;
		bool wels = !r[1] || (ps(p, "SPAIWelsRegistrationNo").Length > 0 && (specWels == null || (pd(p, "SPAIWELSRating") ?? 0m) >= specWels));
		bool gems = !r[2] || (ps(p, "SPAIGemsRegistrationNo").Length > 0 && (specEnergy == null || (pd(p, "SPAIEnergyStarRating") ?? 0m) >= specEnergy));
		bool waterMark = !r[3] || ps(p, "SPAIWaterMarkCertNo").Length > 0;
		if (wels && gems && waterMark) {
			eligible.Add(p);
		} else {
			cutoutFitFloorFail++;
		}
	}
	// Pre-rank: compliance headroom, then availability within lead time, then availability, then finish
	// match, then product code for a stable order. Margin is NOT an input.
	Func<Dictionary<string, object>, decimal> headroom = p =>
		(specWels.HasValue ? (pd(p, "SPAIWELSRating") ?? 0m) - specWels.Value : 0m)
		+ (specEnergy.HasValue ? (pd(p, "SPAIEnergyStarRating") ?? 0m) - specEnergy.Value : 0m);
	Func<Dictionary<string, object>, bool> inTime = p => stockOf(p) >= qty && pi(p, "SPAILeadTimeWeeks") <= LeadTimeLimitWeeks;
	Func<Dictionary<string, object>, bool> inStock = p => stockOf(p) >= qty;
	Func<Dictionary<string, object>, bool> finishMatch = p => string.Equals(ps(p, "SPAIFinish.Name"), finish, StringComparison.OrdinalIgnoreCase);
	eligible.Sort((a, b) => {
		int cmp = headroom(b).CompareTo(headroom(a));
		if (cmp == 0) { cmp = inTime(b).CompareTo(inTime(a)); }
		if (cmp == 0) { cmp = inStock(b).CompareTo(inStock(a)); }
		if (cmp == 0) { cmp = finishMatch(b).CompareTo(finishMatch(a)); }
		if (cmp == 0) { cmp = string.CompareOrdinal(ps(a, "Code"), ps(b, "Code")); }
		return cmp;
	});

		["itemRef"] = str(line, lc["SPAIItemCode"]),

		["specifiedBrand"] = str(line, lc["SPAISpecifiedBrand"]),
		["specifiedModel"] = str(line, lc["SPAISpecifiedModel"]),
		["specifiedFinish"] = finish,
		["quantity"] = qty,
		["cutoutW"] = w, ["cutoutH"] = h, ["cutoutD"] = d,
		["notes"] = str(line, lc["SPAIScheduleNotes"]),
		["regimes"] = new JObject { ["cutout"] = r[0], ["wels"] = r[1], ["gems"] = r[2], ["waterMark"] = r[3] },
		["specifiedProduct"] = sp == null ? (JToken)JValue.CreateNull() : new JObject {
			["productCode"] = ps(sp, "Code"),
			["lifecycleStatus"] = ps(sp, "SPAILifecycleStatus.Name"),

			["leadTimeWeeks"] = pi(sp, "SPAILeadTimeWeeks"),
			["welsRating"] = specWels.HasValue ? (JToken)specWels.Value : JValue.CreateNull(),
			["energyStarRating"] = specEnergy.HasValue ? (JToken)specEnergy.Value : JValue.CreateNull(),

		}
	});

	eventsBySchedule[s].Add(Tuple.Create(e.GetTypedColumnValue<int>(evSeqCol), e.PrimaryColumnValue,
		string.Equals((e.GetTypedColumnValue<string>(evTypeCol) ?? string.Empty).Trim(), "Prototype", StringComparison.OrdinalIgnoreCase),
		e.GetTypedColumnValue<DateTime>(evDateCol)));
}
var emptySchedules = new List<string>();
DateTime firstDate = DateTime.MaxValue, lastDate = DateTime.MinValue;
foreach (Guid s in schedules) {
	eventsBySchedule[s].Sort((a, b) => a.Item1.CompareTo(b.Item1));
	if (eventsBySchedule[s].Count == 0) {
		emptySchedules.Add(scheduleRef[s]);
	}
	foreach (var ev in eventsBySchedule[s]) {
		if (ev.Item4 > DateTime.MinValue && ev.Item4 < firstDate) firstDate = ev.Item4;
		if (ev.Item4 > lastDate) lastDate = ev.Item4;
	}
}
if (emptySchedules.Count > 0) {
	stop("Call-up schedule " + string.Join(", ", emptySchedules) + " of tender " + title
		+ " has no delivery events. The builder's programme must be complete before award.");
	return true;
}

// ---- The supplied lines, and which schedule each is delivered on ----
var lineEsq = new EntitySchemaQuery(uc.EntitySchemaManager, "SPAIScheduleLine");
lineEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
string lnNumCol = lineEsq.AddColumn("SPAILineNumber").Name;
string lnQtyCol = lineEsq.AddColumn("SPAIQuantity").Name;
string lnItemCol = lineEsq.AddColumn("SPAIItemCode").Name;
string lnAltCol = lineEsq.AddColumn("SPAIIsAlternative").Name;
string lnVariantCol = lineEsq.AddColumn("SPAIAlternateVariant").Name;
string lnProductCol = lineEsq.AddColumn("SPAIMatchedProduct").Name;
string lnSellCol = lineEsq.AddColumn("SPAIUnitSell").Name;
string lnStatusCol = lineEsq.AddColumn("SPAILineStatus.Name").Name;
string lnCategoryCol = lineEsq.AddColumn("SPAIMatchedProduct.SPAIProductFamily.SPAICategory").Name;
lineEsq.Filters.Add(lineEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAIOpportunity", opportunityId));
var lines = new List<Entity>();
var lineSchedule = new Dictionary<Guid, Guid>();
var noSchedule = new List<int>();
int notOrdered = 0;
foreach (Entity l in lineEsq.GetEntityCollection(uc)) {
	string st = (l.GetTypedColumnValue<string>(lnStatusCol) ?? string.Empty).Trim();
	bool supplied = st == "Exact match" || st == "Sourced multi-location" || st == "Substitution approved";
	if (!supplied || l.GetTypedColumnValue<Guid>(lnProductCol + "Id") == Guid.Empty || l.GetTypedColumnValue<int>(lnQtyCol) <= 0) {
		notOrdered++;
		continue;
	}
	bool appliance = (l.GetTypedColumnValue<string>(lnCategoryCol) ?? string.Empty)
		.IndexOf("appliance", StringComparison.OrdinalIgnoreCase) >= 0;
	Guid target = Guid.Empty;
	if (schedules.Count == 1) {
		target = schedules[0];
	} else {
		foreach (Guid s in schedules) {
			if (scheduleIsAppliance[s] == appliance) {
				target = s;
				break;
			}
		}
	}
	if (target == Guid.Empty) {
		noSchedule.Add(l.GetTypedColumnValue<int>(lnNumCol));
		continue;
	}
	lines.Add(l);
	lineSchedule[l.PrimaryColumnValue] = target;
}
if (noSchedule.Count > 0) {
	noSchedule.Sort();
	stop("Cannot tell which call-up schedule line " + string.Join(", ", noSchedule) + " is delivered on: "
		+ "the programme's schedule descriptions do not separate appliances from other products. Ask an administrator.");
	return true;
}
if (lines.Count == 0) {
	stop("Tender " + title + " has no approved lines to order.");
	return true;
}
lines.Sort((a, b) => a.GetTypedColumnValue<int>(lnNumCol).CompareTo(b.GetTypedColumnValue<int>(lnNumCol)));

// ---- The indicative sources of those lines ----
var srcEsq = new EntitySchemaQuery(uc.EntitySchemaManager, "SPAILineSource");
srcEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
string srcLineCol = srcEsq.AddColumn("SPAIScheduleLine").Name;
string srcLocCol = srcEsq.AddColumn("SPAILocation").Name;
string srcQtyCol = srcEsq.AddColumn("SPAIQtyAllocated").Name;
string srcTierCol = srcEsq.AddColumn("SPAISourceTier").Name;
string srcPlanCol = srcEsq.AddColumn("SPAISourcePlanType.Name").Name;
srcEsq.Filters.Add(srcEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAIScheduleLine.SPAIOpportunity", opportunityId));
var sources = new List<Entity>();
foreach (Entity s in srcEsq.GetEntityCollection(uc)) {
	string plan = (s.GetTypedColumnValue<string>(srcPlanCol) ?? string.Empty).Trim();
	if (plan == "Committed") {
		stop("Tender " + title + " already has committed stock sources, so it has already been awarded. Nothing is ordered twice.");
		return true;
	}
	if (plan == "Indicative" && lineSchedule.ContainsKey(s.GetTypedColumnValue<Guid>(srcLineCol + "Id"))) {
		sources.Add(s);
	}
}

// ---- Plan every write before making any ----
var productOfLine = new Dictionary<Guid, Guid>();
foreach (Entity l in lines) {
	productOfLine[l.PrimaryColumnValue] = l.GetTypedColumnValue<Guid>(lnProductCol + "Id");
}
var stockNeed = new Dictionary<string, int>();  // "product|location" -> units to encumber
var locationUnits = new Dictionary<Guid, Dictionary<Guid, int>>();  // schedule -> location -> units
var tierUnits = new Dictionary<Guid, Dictionary<Guid, int>>();  // schedule -> tier -> units
var sourcedLines = new HashSet<Guid>();
foreach (Entity s in sources) {
	Guid lineId = s.GetTypedColumnValue<Guid>(srcLineCol + "Id");
	Guid loc = s.GetTypedColumnValue<Guid>(srcLocCol + "Id");
	Guid tier = s.GetTypedColumnValue<Guid>(srcTierCol + "Id");
	int qty = s.GetTypedColumnValue<int>(srcQtyCol);
	if (qty <= 0) {
		continue;
	}
	sourcedLines.Add(lineId);
	Guid sched = lineSchedule[lineId];
	if (loc != Guid.Empty) {
		string key = productOfLine[lineId] + "|" + loc;
		stockNeed[key] = (stockNeed.ContainsKey(key) ? stockNeed[key] : 0) + qty;
		if (!locationUnits.ContainsKey(sched)) locationUnits[sched] = new Dictionary<Guid, int>();
		locationUnits[sched][loc] = (locationUnits[sched].ContainsKey(loc) ? locationUnits[sched][loc] : 0) + qty;
	}
	if (tier != Guid.Empty) {
		if (!tierUnits.ContainsKey(sched)) tierUnits[sched] = new Dictionary<Guid, int>();
		tierUnits[sched][tier] = (tierUnits[sched].ContainsKey(tier) ? tierUnits[sched][tier] : 0) + qty;
	}
}
Func<Dictionary<Guid, Dictionary<Guid, int>>, Guid, Guid> dominant = (bySchedule, sched) => {
	Guid best = Guid.Empty;
	int most = 0;
	if (bySchedule.ContainsKey(sched)) {
		foreach (var kv in bySchedule[sched]) {
			if (kv.Value > most) {
				most = kv.Value;
				best = kv.Key;
			}
		}
	}
	return best;
};

// Quantity per (line, event): prototype events first, then an even spread, remainder on the last event.
var callUps = new List<Tuple<Entity, Guid, int>>();  // line, event id, qty
var eventsUsedByLine = new Dictionary<Guid, int>();
foreach (Entity l in lines) {
	var evs = eventsBySchedule[lineSchedule[l.PrimaryColumnValue]];
	int remaining = l.GetTypedColumnValue<int>(lnQtyCol);
	var rollout = new List<Guid>();
	int planned = 0;
	foreach (var ev in evs) {
		if (ev.Item3) {
			int take = Math.Min(remaining, prototypeQty);
			if (take > 0) {
				callUps.Add(Tuple.Create(l, ev.Item2, take));
				remaining -= take;
				planned++;
			}
		} else {
			rollout.Add(ev.Item2);
		}
	}
	if (remaining > 0 && rollout.Count == 0) {
		// A schedule of prototype events only: the rest goes on its last event.
		rollout.Add(evs[evs.Count - 1].Item2);
	}
	if (remaining > 0) {
		int each = remaining / rollout.Count;
		int spare = remaining - each * rollout.Count;
		for (int i = 0; i < rollout.Count; i++) {
			int take = each + (i == rollout.Count - 1 ? spare : 0);
			if (take > 0) {
				callUps.Add(Tuple.Create(l, rollout[i], take));
				planned++;
			}
		}
	}
	eventsUsedByLine[l.PrimaryColumnValue] = planned;
}
var linesAtEvent = new Dictionary<Guid, HashSet<Guid>>();
var valueAtEvent = new Dictionary<Guid, decimal>();
foreach (var cu in callUps) {
	if (!linesAtEvent.ContainsKey(cu.Item2)) {
		linesAtEvent[cu.Item2] = new HashSet<Guid>();
		valueAtEvent[cu.Item2] = 0m;
	}
	linesAtEvent[cu.Item2].Add(cu.Item1.PrimaryColumnValue);
	valueAtEvent[cu.Item2] += cu.Item3 * cu.Item1.GetTypedColumnValue<decimal>(lnSellCol);
}

// 2. Commit the sources, then encumber stock
EntitySchema sourceSchema = uc.EntitySchemaManager.GetInstanceByName("SPAILineSource");

	Entity row = sourceSchema.CreateEntity(uc);
	if (row.FetchFromDB(s.PrimaryColumnValue)) {
		row.SetColumnValue("SPAISourcePlanTypeId", planType["Committed"]);
		row.Save(false);
		committed++;
	}

EntitySchema stockSchema = uc.EntitySchemaManager.GetInstanceByName("SPAIStockPosition");
var locationsTouched = new HashSet<string>();
foreach (var kv in stockNeed) {
	string[] ids = kv.Key.Split('|');
	var posEsq = new EntitySchemaQuery(uc.EntitySchemaManager, "SPAIStockPosition");
	posEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
	string allocCol = posEsq.AddColumn("SPAIQtyAllocated").Name;
	string availCol = posEsq.AddColumn("SPAIQtyAvailable").Name;
	posEsq.Filters.Add(posEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAIProduct", new Guid(ids[0])));
	posEsq.Filters.Add(posEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAILocation", new Guid(ids[1])));
	foreach (Entity position in posEsq.GetEntityCollection(uc)) {
		Entity update = stockSchema.CreateEntity(uc);
		if (update.FetchFromDB(position.PrimaryColumnValue)) {
			update.SetColumnValue("SPAIQtyAllocated", position.GetTypedColumnValue<int>(allocCol) + kv.Value);
			update.SetColumnValue("SPAIQtyAvailable", Math.Max(0, position.GetTypedColumnValue<int>(availCol) - kv.Value));
			update.Save(false);
			locationsTouched.Add(ids[1]);
		}
		break;
	}
}
Set("CommittedCount", committed);

// 3. One sub-PO per delivery event, numbered from the blanket
EntitySchema deliverySchema = uc.EntitySchemaManager.GetInstanceByName("SPAIDelivery");
var subPoByEvent = new Dictionary<Guid, Guid>();
int poSeq = 0;
foreach (Guid s in schedules) {
	Guid loc = dominant(locationUnits, s);
	Guid tier = dominant(tierUnits, s);
	foreach (var ev in eventsBySchedule[s]) {
		poSeq++;
		Entity po = orderSchema.CreateEntity(uc);
		orderBasics(po);
		string subRef = string.Format("SPO-{0}-{1:00}", poDigits, poSeq);
		po.SetColumnValue("Number", subRef);
		po.SetColumnValue("SPAIPurchaseOrderNo", subRef);
		po.SetColumnValue("SPAIOrderTypeId", orderTypeCallOff);
		po.SetColumnValue("SPAIBlanketOrderId", blanketId);
		po.SetColumnValue("SPAIDeliveryEventId", ev.Item2);
		if (ev.Item4 > DateTime.MinValue) {
			po.SetColumnValue("SPAITargetDate", ev.Item4);
		}
		if (loc != Guid.Empty) {
			po.SetColumnValue("SPAIPrimaryLocationId", loc);
		}
		if (tier != Guid.Empty) {
			po.SetColumnValue("SPAISourceTierId", tier);
		}
		po.Save(false);
		subPoByEvent[ev.Item2] = po.PrimaryColumnValue;

		Entity delivery = deliverySchema.CreateEntity(uc);
		delivery.SetDefColumnValues();
		delivery.SetColumnValue("SPAIOrderId", po.PrimaryColumnValue);
		if (loc != Guid.Empty) {
			delivery.SetColumnValue("SPAIFromLocationId", loc);
		}
		if (ev.Item4 > DateTime.MinValue) {
			delivery.SetColumnValue("SPAIScheduledOn", ev.Item4);
		}
		delivery.SetColumnValue("SPAIStatusId", deliveryStatus["Scheduled"]);
		delivery.SetColumnValue("SPAILineCount", linesAtEvent.ContainsKey(ev.Item2) ? linesAtEvent[ev.Item2].Count : 0);
		delivery.SetColumnValue("SPAIValue", valueAtEvent.ContainsKey(ev.Item2) ? valueAtEvent[ev.Item2] : 0m);
		delivery.Save(false);
	}
}
Set("SubPoCount", subPoByEvent.Count);

// 4. Call-up lines and order products
EntitySchema callUpSchema = uc.EntitySchemaManager.GetInstanceByName("SPAICallUpLine");
