// ---- The tender ----

Guid parsedId = Guid.Empty;

oppEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
string oppTitleCol = oppEsq.AddColumn("Title").Name;
string oppStatusCol = oppEsq.AddColumn("SPAIAdjudicationStatus.Name").Name;
string oppAccountCol = oppEsq.AddColumn("Account").Name;

	oppEsq.Filters.Add(oppEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "Id", parsedId));
} else {
	var either = new EntitySchemaQueryFilterCollection(oppEsq, Terrasoft.Common.LogicalOperationStrict.Or);
	either.Add(oppEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "Title", reference));
	either.Add(oppEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAITenderCode", reference));

foreach (Entity o in oppEsq.GetEntityCollection(uc)) {
	found++;
	opportunityId = o.PrimaryColumnValue;
	title = (o.GetTypedColumnValue<string>(oppTitleCol) ?? string.Empty).Trim();
	tenderStatus = (o.GetTypedColumnValue<string>(oppStatusCol) ?? string.Empty).Trim();
	accountId = o.GetTypedColumnValue<Guid>(oppAccountCol + "Id");
}
if (found == 0 && parsedId == Guid.Empty) {
	// Same name written differently (punctuation, spacing, case): use it only when exactly one tender matches.
	Func<string, string> normalise = value => {
		var kept = new System.Text.StringBuilder();
		foreach (char c in value ?? string.Empty) {
			if (char.IsLetterOrDigit(c)) {
				kept.Append(char.ToLowerInvariant(c));
			}
		}
		return kept.ToString();
	};
	string wanted = normalise(reference);
	string anchor = string.Empty;
	foreach (string word in reference.Split(new[] { ' ', ',', '-', '/', '.', '(', ')' }, StringSplitOptions.RemoveEmptyEntries)) {
		if (word.Length > anchor.Length) {
			anchor = word;
		}
	}
	if (wanted.Length > 0 && anchor.Length > 0) {
		var nearEsq = new EntitySchemaQuery(uc.EntitySchemaManager, "Opportunity");
		nearEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
		string nTitleCol = nearEsq.AddColumn("Title").Name;
		string nCodeCol = nearEsq.AddColumn("SPAITenderCode").Name;
		string nStatusCol = nearEsq.AddColumn("SPAIAdjudicationStatus.Name").Name;
		string nAccountCol = nearEsq.AddColumn("Account").Name;
		var anyText = new EntitySchemaQueryFilterCollection(nearEsq, Terrasoft.Common.LogicalOperationStrict.Or);
		anyText.Add(nearEsq.CreateFilterWithParameters(FilterComparisonType.Contain, "Title", anchor));
		anyText.Add(nearEsq.CreateFilterWithParameters(FilterComparisonType.Contain, "SPAITenderCode", anchor));
		nearEsq.Filters.Add(anyText);
		foreach (Entity o in nearEsq.GetEntityCollection(uc)) {
			if (normalise(o.GetTypedColumnValue<string>(nTitleCol)) == wanted || normalise(o.GetTypedColumnValue<string>(nCodeCol)) == wanted) {
				found++;
				opportunityId = o.PrimaryColumnValue;
				title = (o.GetTypedColumnValue<string>(nTitleCol) ?? string.Empty).Trim();
				tenderStatus = (o.GetTypedColumnValue<string>(nStatusCol) ?? string.Empty).Trim();
				accountId = o.GetTypedColumnValue<Guid>(nAccountCol + "Id");
			}
		}
	}
}

	stop("No tender is titled or coded exactly \"" + reference + "\". Use the tender name the intake reported.");
	return true;
}

// ---- Plan every write before making any ----
var productOfLine = new Dictionary<Guid, Guid>();
foreach (Entity l in lines) {
	productOfLine[l.PrimaryColumnValue] = l.GetTypedColumnValue<Guid>(lnProductCol + "Id");
}
var stockNeed = new Dictionary<string, int>();  // "product|location" -> units to encumber
var locationUnits = new Dictionary<Guid, Dictionary<Guid, int>>();  // schedule -> location -> units
var tierUnits = new Dictionary<Guid, Dictionary<Guid, int>>();  // schedule -> tier -> units
var sourcedLines = new HashSet<Guid>();
foreach (Entity s in sources) {
	Guid lineId = s.GetTypedColumnValue<Guid>(srcLineCol + "Id");
	Guid loc = s.GetTypedColumnValue<Guid>(srcLocCol + "Id");
	Guid tier = s.GetTypedColumnValue<Guid>(srcTierCol + "Id");
	int qty = s.GetTypedColumnValue<int>(srcQtyCol);
	if (qty <= 0) {
		continue;
	}
	sourcedLines.Add(lineId);
	Guid sched = lineSchedule[lineId];
	if (loc != Guid.Empty) {
		string key = productOfLine[lineId] + "|" + loc;
		stockNeed[key] = (stockNeed.ContainsKey(key) ? stockNeed[key] : 0) + qty;
		if (!locationUnits.ContainsKey(sched)) locationUnits[sched] = new Dictionary<Guid, int>();
		locationUnits[sched][loc] = (locationUnits[sched].ContainsKey(loc) ? locationUnits[sched][loc] : 0) + qty;
	}
	if (tier != Guid.Empty) {
		if (!tierUnits.ContainsKey(sched)) tierUnits[sched] = new Dictionary<Guid, int>();
		tierUnits[sched][tier] = (tierUnits[sched].ContainsKey(tier) ? tierUnits[sched][tier] : 0) + qty;
	}
}
Func<Dictionary<Guid, Dictionary<Guid, int>>, Guid, Guid> dominant = (bySchedule, sched) => {
	Guid best = Guid.Empty;
	int most = 0;
	if (bySchedule.ContainsKey(sched)) {
		foreach (var kv in bySchedule[sched]) {
			if (kv.Value > most) {
				most = kv.Value;
				best = kv.Key;
			}
		}
	}
	return best;
};

// Quantity per (line, event): prototype events first, then an even spread, remainder on the last event.
var callUps = new List<Tuple<Entity, Guid, int>>();  // line, event id, qty
var eventsUsedByLine = new Dictionary<Guid, int>();
foreach (Entity l in lines) {
	var evs = eventsBySchedule[lineSchedule[l.PrimaryColumnValue]];
	int remaining = l.GetTypedColumnValue<int>(lnQtyCol);
	var rollout = new List<Guid>();
	int planned = 0;
	foreach (var ev in evs) {
		if (ev.Item3) {
			int take = Math.Min(remaining, prototypeQty);
			if (take > 0) {
				callUps.Add(Tuple.Create(l, ev.Item2, take));
				remaining -= take;
				planned++;
			}
		} else {
			rollout.Add(ev.Item2);
		}
	}
	if (remaining > 0 && rollout.Count == 0) {
		// A schedule of prototype events only: the rest goes on its last event.
		rollout.Add(evs[evs.Count - 1].Item2);
	}
	if (remaining > 0) {
		int each = remaining / rollout.Count;
		int spare = remaining - each * rollout.Count;
		for (int i = 0; i < rollout.Count; i++) {
			int take = each + (i == rollout.Count - 1 ? spare : 0);
			if (take > 0) {
				callUps.Add(Tuple.Create(l, rollout[i], take));
				planned++;
			}
		}
	}
	eventsUsedByLine[l.PrimaryColumnValue] = planned;
}
var linesAtEvent = new Dictionary<Guid, HashSet<Guid>>();
var valueAtEvent = new Dictionary<Guid, decimal>();
foreach (var cu in callUps) {
	if (!linesAtEvent.ContainsKey(cu.Item2)) {
		linesAtEvent[cu.Item2] = new HashSet<Guid>();
		valueAtEvent[cu.Item2] = 0m;
	}
	linesAtEvent[cu.Item2].Add(cu.Item1.PrimaryColumnValue);
	valueAtEvent[cu.Item2] += cu.Item3 * cu.Item1.GetTypedColumnValue<decimal>(lnSellCol);
}

EntitySchema stockSchema = uc.EntitySchemaManager.GetInstanceByName("SPAIStockPosition");
var locationsTouched = new HashSet<string>();
foreach (var kv in stockNeed) {
	string[] ids = kv.Key.Split('|');
	var posEsq = new EntitySchemaQuery(uc.EntitySchemaManager, "SPAIStockPosition");
	posEsq.PrimaryQueryColumn.IsAlwaysSelect = true;
	string allocCol = posEsq.AddColumn("SPAIQtyAllocated").Name;
	string availCol = posEsq.AddColumn("SPAIQtyAvailable").Name;
	posEsq.Filters.Add(posEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAIProduct", new Guid(ids[0])));
	posEsq.Filters.Add(posEsq.CreateFilterWithParameters(FilterComparisonType.Equal, "SPAILocation", new Guid(ids[1])));
	foreach (Entity position in posEsq.GetEntityCollection(uc)) {
		Entity update = stockSchema.CreateEntity(uc);
		if (update.FetchFromDB(position.PrimaryColumnValue)) {
			update.SetColumnValue("SPAIQtyAllocated", position.GetTypedColumnValue<int>(allocCol) + kv.Value);
			update.SetColumnValue("SPAIQtyAvailable", Math.Max(0, position.GetTypedColumnValue<int>(availCol) - kv.Value));
			update.Save(false);
			locationsTouched.Add(ids[1]);
		}
		break;
	}
}
Set("CommittedCount", committed);

// 3. One sub-PO per delivery event, numbered from the blanket
EntitySchema deliverySchema = uc.EntitySchemaManager.GetInstanceByName("SPAIDelivery");
var subPoByEvent = new Dictionary<Guid, Guid>();
int poSeq = 0;
foreach (Guid s in schedules) {
	Guid loc = dominant(locationUnits, s);
	Guid tier = dominant(tierUnits, s);
	foreach (var ev in eventsBySchedule[s]) {
		poSeq++;
		Entity po = orderSchema.CreateEntity(uc);
		orderBasics(po);
		string subRef = string.Format("SPO-{0}-{1:00}", poDigits, poSeq);
		po.SetColumnValue("Number", subRef);
		po.SetColumnValue("SPAIPurchaseOrderNo", subRef);
		po.SetColumnValue("SPAIOrderTypeId", orderTypeCallOff);
		po.SetColumnValue("SPAIBlanketOrderId", blanketId);
		po.SetColumnValue("SPAIDeliveryEventId", ev.Item2);
		if (ev.Item4 > DateTime.MinValue) {
			po.SetColumnValue("SPAITargetDate", ev.Item4);
		}
		if (loc != Guid.Empty) {
			po.SetColumnValue("SPAIPrimaryLocationId", loc);
		}
		if (tier != Guid.Empty) {
			po.SetColumnValue("SPAISourceTierId", tier);
		}
		po.Save(false);
		subPoByEvent[ev.Item2] = po.PrimaryColumnValue;

		Entity delivery = deliverySchema.CreateEntity(uc);
		delivery.SetDefColumnValues();
		delivery.SetColumnValue("SPAIOrderId", po.PrimaryColumnValue);
		if (loc != Guid.Empty) {
			delivery.SetColumnValue("SPAIFromLocationId", loc);
		}
		if (ev.Item4 > DateTime.MinValue) {
			delivery.SetColumnValue("SPAIScheduledOn", ev.Item4);
		}
		delivery.SetColumnValue("SPAIStatusId", deliveryStatus["Scheduled"]);
		delivery.SetColumnValue("SPAILineCount", linesAtEvent.ContainsKey(ev.Item2) ? linesAtEvent[ev.Item2].Count : 0);
		delivery.SetColumnValue("SPAIValue", valueAtEvent.ContainsKey(ev.Item2) ? valueAtEvent[ev.Item2] : 0m);
		delivery.Save(false);
	}
}
Set("SubPoCount", subPoByEvent.Count);

// 4. Call-up lines and order products
EntitySchema callUpSchema = uc.EntitySchemaManager.GetInstanceByName("SPAICallUpLine");

int callUpCount = 0;
foreach (var cu in callUps) {
	Entity l = cu.Item1;
	Entity cul = callUpSchema.CreateEntity(uc);
	cul.SetDefColumnValues();
	cul.SetColumnValue("SPAIScheduleLineId", l.PrimaryColumnValue);
	cul.SetColumnValue("SPAIDeliveryEventId", cu.Item2);
	cul.SetColumnValue("SPAISubPOId", subPoByEvent[cu.Item2]);
	cul.SetColumnValue("SPAIQtyRequired", cu.Item3);
	cul.SetColumnValue("SPAIQtyDelivered", 0);
	cul.SetColumnValue("SPAIStatusId", deliveryStatus["Scheduled"]);
	cul.Save(false);
	callUpCount++;

	Entity op = orderProductSchema.CreateEntity(uc);
	op.SetDefColumnValues();
	op.SetColumnValue("OrderId", subPoByEvent[cu.Item2]);
	op.SetColumnValue("ProductId", productOfLine[l.PrimaryColumnValue]);
	op.SetColumnValue("Quantity", (decimal)cu.Item3);
	op.SetColumnValue("Price", l.GetTypedColumnValue<decimal>(lnSellCol));
	op.SetColumnValue("SPAIScheduleLineId", l.PrimaryColumnValue);
	op.SetColumnValue("SPAIItemCode", l.GetTypedColumnValue<string>(lnItemCol));
	op.SetColumnValue("SPAIIsAlternative", l.GetTypedColumnValue<bool>(lnAltCol));
	op.SetColumnValue("SPAIAlternateVariant", l.GetTypedColumnValue<string>(lnVariantCol));
	op.Save(false);  // SPAIItemIdentityEntityEventListener derives SPAIDisplayRef on save
}
Set("CallUpLineCount", callUpCount);

// 5. The ledger: one row per line, one for the award
EntitySchema ledgerSchema = uc.EntitySchemaManager.GetInstanceByName("SPAIDecisionLedger");
var unsourced = new List<int>();
foreach (Entity l in lines) {
	int n = l.GetTypedColumnValue<int>(lnNumCol);
	bool hasStock = sourcedLines.Contains(l.PrimaryColumnValue);
	if (!hasStock) {
		unsourced.Add(n);
	}
	Entity ledger = ledgerSchema.CreateEntity(uc);
	ledger.SetDefColumnValues();
	ledger.SetColumnValue("SPAIOpportunityId", opportunityId);
	ledger.SetColumnValue("SPAIScheduleLineId", l.PrimaryColumnValue);
	ledger.SetColumnValue("SPAIDecisionTypeId", decisionType["Sourcing allocation"]);
	ledger.SetColumnValue("SPAIActor", actor);
	ledger.SetColumnValue("SPAIComplianceChecks", string.Format(
		"award under {0}: {1} units across {2} call-up line(s) on schedule {3}; {4}",
		poNumber, l.GetTypedColumnValue<int>(lnQtyCol), eventsUsedByLine[l.PrimaryColumnValue],
		scheduleRef[lineSchedule[l.PrimaryColumnValue]],
		hasStock ? "stock sources converted to Committed and encumbered now" : "no stock source: to be supplied from inbound stock"));
	ledger.SetColumnValue("SPAIPriorValue", "Indicative source plan");
	ledger.SetColumnValue("SPAINewValue", "Committed, call-up raised");
	ledger.SetColumnValue("SPAISequence", n);
	ledger.SetColumnValue("SPAIOccurredOn", now);
	ledger.Save(false);
}
Entity award = ledgerSchema.CreateEntity(uc);
award.SetDefColumnValues();
award.SetColumnValue("SPAIOpportunityId", opportunityId);
award.SetColumnValue("SPAIDecisionTypeId", decisionType["Human override"]);
award.SetColumnValue("SPAIActor", actor);
award.SetColumnValue("SPAIComplianceChecks", string.Format(
	"Gate 2: award approved in the chat after Tool Confirmation. Blanket PO {0}: {1} line(s) ordered, {2} sub-PO(s) and deliveries, {3} call-up line(s), {4} source(s) committed and stock reserved{5}",
	poNumber, lines.Count, subPoByEvent.Count, callUpCount, committed,
	note.Length > 0 ? ". Note from the person: " + note : string.Empty));
award.SetColumnValue("SPAIPriorValue", "Submitted");
award.SetColumnValue("SPAINewValue", "Awarded | Closed won | fulfilment raised");
award.SetColumnValue("SPAISequence", 0);
award.SetColumnValue("SPAIOccurredOn", now);
award.Save(false);

// 6. The Opportunity: Awarded, Closed won, Gate 2 stamped. Last, so the screen flips once everything exists.
Entity opportunity = uc.EntitySchemaManager.GetInstanceByName("Opportunity").CreateEntity(uc);
if (opportunity.FetchFromDB(opportunityId)) {
	opportunity.SetColumnValue("SPAIAdjudicationStatusId", tenderStatusByName["Awarded"]);
	opportunity.SetColumnValue("StageId", stageClosedWon);
	opportunity.SetColumnValue("SPAIGate2ApprovedOn", now);
	opportunity.SetColumnValue("SPAIGate2ApprovedById", uc.CurrentUser.ContactId);
	opportunity.Save(false);
}

	unsourced.Sort();
